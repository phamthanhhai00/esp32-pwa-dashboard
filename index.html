/*
 * CODE ESP32 BLUETOOTH BLE (Sử dụng với Arduino IDE)
 */
#include <BLEDevice.h>
#include <BLEServer.h>
#include <BLEUtils.h>
#include <BLE2902.h>

#define SERVICE_UUID        "4fa0c101-0001-4000-8000-000000000000"
#define CHARACTERISTIC_UUID "4fa0c101-0002-4000-8000-000000000000"

BLECharacteristic *pCharacteristic;
bool deviceConnected = false;

class MyServerCallbacks: public BLEServerCallbacks {
    void onConnect(BLEServer* pServer) { deviceConnected = true; }
    void onDisconnect(BLEServer* pServer) { deviceConnected = false; BLEDevice::startAdvertising(); }
};

class MyCallbacks: public BLECharacteristicCallbacks {
    void onWrite(BLECharacteristic *pCharacteristic) {
      String rxValue = pCharacteristic->getValue();
      if (rxValue.length() > 0) {
        Serial.print("Lệnh từ Web Dashboard: ");
        Serial.println(rxValue);
      }
    }
};

void setup() {
  Serial.begin(115200);
  BLEDevice::init("ESP32_NEXUS");
  BLEServer *pServer = BLEDevice::createServer();
  pServer->setCallbacks(new MyServerCallbacks());
  
  BLEService *pService = pServer->createService(SERVICE_UUID);
  pCharacteristic = pService->createCharacteristic(
                      CHARACTERISTIC_UUID,
                      BLECharacteristic::PROPERTY_READ   |
                      BLECharacteristic::PROPERTY_WRITE  |
                      BLECharacteristic::PROPERTY_NOTIFY
                    );
  pCharacteristic->addDescriptor(new BLE2902());
  pCharacteristic->setCallbacks(new MyCallbacks());
  pService->start();
  
  BLEAdvertising *pAdvertising = BLEDevice::getAdvertising();
  pAdvertising->addServiceUUID(SERVICE_UUID);
  pAdvertising->start();
  Serial.println("ESP32 BLE Sẵn sàng kết nối!");
}

void loop() {
  if (deviceConnected) {
    // Gửi dữ liệu Telemetry cảm biến JSON định kỳ 2 giây
    float t = 28.5 + random(-10, 10)/10.0;
    float h = 65.0 + random(-20, 20)/10.0;
    String jsonStr = "{\"temp\":" + String(t) + ",\"hum\":" + String(h) + ",\"volt\":3.8,\"rssi\":-55}";
    pCharacteristic->setValue(jsonStr.c_str());
    pCharacteristic->notify();
    delay(2000);
  }
}
                
/*
 * CODE ESP32 WIFI AP LOCAL SERVER (Sử dụng với Arduino IDE)
 */
#include <WiFi.h>
#include <WebServer.h>

WebServer server(80);

void handleData() {
  float t = 29.1 + random(-10, 10)/10.0;
  float h = 70.0 + random(-20, 20)/10.0;
  String jsonStr = "{\"temp\":" + String(t) + ",\"hum\":" + String(h) + ",\"volt\":4.1,\"rssi\":-42}";
  
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "application/json", jsonStr);
}

void handleControl() {
  if (server.hasArg("cmd")) {
    String cmd = server.arg("cmd");
    Serial.println("Lệnh Web HTTP: " + cmd);
  }
  server.sendHeader("Access-Control-Allow-Origin", "*");
  server.send(200, "text/plain", "OK");
}

void setup() {
  Serial.begin(115200);
  
  // Phát mạng WiFi AP riêng biệt
  WiFi.softAP("ESP32_NEXUS_AP", "12345678");
  Serial.print("Web Server IP: ");
  Serial.println(WiFi.softAPIP()); // Mặc định: 192.168.4.1

  server.on("/api/data", HTTP_GET, handleData);
  server.on("/api/control", HTTP_GET, handleControl);
  server.enableCORS(true);
  server.begin();
}

void loop() {
  server.handleClient();
}

Thêm App Vào Màn Hình Chính

Dành cho Android (Chrome)

  1. Mở file HTML này trong trình duyệt **Google Chrome**.
  2. Bấm dấu **3 chấm (Menu)** ở góc trên bên phải.
  3. Chọn **"Thêm vào màn hình chính" (Add to Home screen)**.
  4. Mở App từ biểu tượng mới tạo ngoài màn hình!

Dành cho iOS iPhone/iPad (Safari)

  1. Mở file HTML bằng trình duyệt **Safari**.
  2. Bấm nút **Chia sẻ (Share icon)** ở thanh công cụ dưới.
  3. Cuộn xuống và chọn **"Thêm vào Màn hình chính" (Add to Home Screen)**.
Lưu ý Bluetooth trên iOS: Do giới hạn của Apple, Web Bluetooth chỉ chạy mượt trên Android/Chrome hoặc WebBLE App trên iOS. Nếu dùng iPhone chuẩn, khuyến khích chuyển sang kết nối **WiFi AP**.
